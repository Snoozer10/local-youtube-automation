4.4 Automated Dataset Sanitization Script (Python Normalizer for Google Flow)import json
import re

def sanitize_flow_dataset(input_filepath: str, output_filepath: str) -> None:
    with open(input_filepath, 'r', encoding='utf-8') as f:
        data = json.load(f)
        
    for frame in data:
        prompt = frame.get('socratic_prompt', '')
        
        # 1. Purge recursive regex replacement artifacts
        prompt = re.sub(r'(clean 2D vector graphic lines with zero text\s*){2,}', 'clean 2D vector schematics ', prompt)
        prompt = re.sub(r'(blank dark glass monitor without display\s*){2,}', 'digital telemetry display ', prompt)
        prompt = re.sub(r'(ornate blank brass stamping tool without lettering\s*){2,}', 'stylized wax seal emblem ', prompt)
        prompt = re.sub(r'circuit blank unmarked wooden board with zero writing', 'circuit schematic board', prompt)
        prompt = re.sub(r'open retro blank closed book with unmarked plain cover', 'open technical reference ledger', prompt)
        prompt = re.sub(r'diagram diagram', 'diagram', prompt)
        
        # 2. Eliminate camera lens conflicts (replace 24mm with orthographic)
        prompt = prompt.replace('24mm wide-angle lens', 'orthographic flat 2D animation plane, zero barrel distortion')
        
        # 3. Enforce 16:9 Long-Form safe zones with 10% pan/zoom bleed padding
        if 'coordinates X: 180 to 1740' not in prompt:
            prompt = re.sub(
                r'clean centered 16:9 widescreen framing,?\s*',
                'clean 16:9 widescreen framing bounded within coordinates X: 180 to 1740 and Y: 90 to 980 with 10% bleed padding, ',
                prompt
            )
            
        # 4. Cleanse negative prompt of mutually destructive tokens and normalize delimiters
        neg = frame.get('negative_prompt', '')
        neg = re.sub(r'\bno 3px\b,?', '', neg)
        neg = re.sub(r'\bno vector\b,?', '', neg)
        neg = re.sub(r'\bno cel-shading\b,?', '', neg)
        neg = re.sub(r',\s*,+', ', ', neg)
        neg = re.sub(r'^\s*,\s*|\s*,\s*$', '', neg)
        
        frame['socratic_prompt'] = prompt
        frame['negative_prompt'] = neg.strip()
        
    with open(output_filepath, 'w', encoding='utf-8') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)
        
    print(f"Sanitized {len(data)} frames successfully. Output saved to {output_filepath}")

if __name__ == "__main__":
    sanitize_flow_dataset("frames_data.json", "frames_data_sanitized.json")
